cmake_minimum_required(VERSION 3.24)
project(StrataCopy VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_INSTALL_BINDIR ".")
set(CMAKE_INSTALL_LIBDIR ".")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(STRATACOPY_ENABLE_UI_TEST_HOOKS "Include timed graphical QA controls (never enabled in a normal release)" OFF)
if(NOT WIN32)
  message(FATAL_ERROR "StrataCopy requires Windows and the system Robocopy executable.")
endif()
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 ShaderTools Network Sql Concurrent Test)
qt_standard_project_setup(REQUIRES 6.7)
add_library(stratacopy_engine STATIC
  src/Core/RobocopyProfile.cpp src/Core/RobocopyCommandCompiler.cpp
  src/Core/DeletionSupervisor.cpp
  src/Core/UpdateChecker.cpp
  src/Core/MultiSourceBatcher.cpp src/Core/StdoutPipeParser.cpp
  src/Core/KernelTelemetryPoller.cpp src/Core/FastPreScanner.cpp
  src/Core/RobocopyProcessSupervisor.cpp src/Verification/PostCopyHashVerifier.cpp
  src/Persistence/SqliteQueueStore.cpp src/Platform/PrivilegeEscalator.cpp
  src/Platform/DriveHardwareDetector.cpp src/Platform/ShellContextMenuRegistrar.cpp
  src/UI/Models/FileTransferListModel.cpp src/UI/Models/TelemetryHistoryModel.cpp
  src/UI/Models/PresetListModel.cpp src/UI/ViewModels/MainCoordinatorViewModel.cpp
  third_party/xxhash/xxhash.c third_party/blake3/blake3.c
  third_party/blake3/blake3_dispatch.c third_party/blake3/blake3_portable.c)
target_include_directories(stratacopy_engine PUBLIC src third_party/xxhash third_party/blake3)
target_compile_definitions(stratacopy_engine PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
  BLAKE3_NO_SSE2 BLAKE3_NO_SSE41 BLAKE3_NO_AVX2 BLAKE3_NO_AVX512)
target_link_libraries(stratacopy_engine PUBLIC Qt6::Core Qt6::Gui Qt6::Network Qt6::Sql Qt6::Concurrent advapi32 shell32 ole32 uuid powrprof)
qt_add_executable(StrataCopy WIN32 src/main.cpp src/Platform/Win32WindowEffects.cpp resources/StrataCopy.rc)
qt_add_executable(StrataCopyExplorer WIN32 src/Platform/ExplorerPasteAgent.cpp resources/StrataCopy.rc)
target_compile_definitions(StrataCopyExplorer PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(StrataCopyExplorer PRIVATE Qt6::Core ole32 oleaut32 uuid shell32 user32 advapi32)
target_compile_definitions(StrataCopyExplorer PRIVATE STRATACOPY_NATIVE_DLL_NAME="StrataCopyNative-${PROJECT_VERSION}-p5.dll")
target_compile_definitions(stratacopy_engine PRIVATE STRATACOPY_NATIVE_DLL_NAME="StrataCopyNative-${PROJECT_VERSION}-p5.dll")
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(StrataCopyNative SHARED src/Platform/ExplorerNativeHook.cpp)
set_target_properties(StrataCopyNative PROPERTIES PREFIX "" AUTOMOC OFF)
target_compile_definitions(StrataCopyNative PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(StrataCopyNative PRIVATE minhook ole32 shell32 uuid user32)
if(MINGW)
  target_link_options(StrataCopyNative PRIVATE -static-libgcc -static-libstdc++ -Wl,-Bstatic,--whole-archive -lwinpthread -Wl,--no-whole-archive,-Bdynamic)
endif()
if(STRATACOPY_ENABLE_UI_TEST_HOOKS)
  target_compile_definitions(StrataCopy PRIVATE STRATACOPY_UI_TEST_HOOKS)
endif()
target_link_libraries(StrataCopy PRIVATE stratacopy_engine Qt6::Quick Qt6::QuickControls2 Qt6::GuiPrivate Qt6::Widgets dwmapi psapi)
set_source_files_properties(src/UI/qml/Theme/LiquidGlassTheme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
file(GLOB_RECURSE APP_QML CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} src/UI/qml/*.qml)
foreach(qml IN LISTS APP_QML)
  get_filename_component(alias ${qml} NAME)
  set_source_files_properties(${qml} PROPERTIES QT_RESOURCE_ALIAS ${alias})
endforeach()
qt_add_qml_module(StrataCopy URI StrataCopy VERSION 1.0 QML_FILES ${APP_QML})
qt_add_resources(StrataCopy brand_assets PREFIX / FILES resources/icons/stratacopy.svg)
qt_add_shaders(StrataCopy strata_shaders PREFIX /shaders BASE resources/shaders FILES
  resources/shaders/liquid_glass_card.frag resources/shaders/liquid_plasma_bar.frag)
if(MSVC)
  target_compile_options(stratacopy_engine PRIVATE /O2 /utf-8 /permissive- /W4)
  target_compile_options(StrataCopy PRIVATE /O2 /utf-8 /permissive-)
else()
  target_compile_options(stratacopy_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(stratacopy_tests tests/EngineTests.cpp)
  target_link_libraries(stratacopy_tests PRIVATE stratacopy_engine Qt6::Test psapi)
  add_test(NAME engine COMMAND stratacopy_tests)
  qt_add_executable(stratacopy_native_tests tests/NativeHookTests.cpp)
  target_compile_definitions(stratacopy_native_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(stratacopy_native_tests PRIVATE Qt6::Test shell32 ole32 uuid psapi)
  target_include_directories(stratacopy_native_tests PRIVATE src)
  add_dependencies(stratacopy_native_tests StrataCopyNative StrataCopy)
  add_test(NAME native_hook COMMAND stratacopy_native_tests)
endif()
install(TARGETS StrataCopy RUNTIME DESTINATION .)
install(TARGETS StrataCopyExplorer RUNTIME DESTINATION .)
install(FILES $<TARGET_FILE:StrataCopyNative> DESTINATION . RENAME "StrataCopyNative-${PROJECT_VERSION}-p5.dll")
install(FILES README.md RELEASE_NOTES.md MEMORY_REVIEW.md THIRD_PARTY_NOTICES.md VALIDATION.md DESTINATION .)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/test-results.txt DESTINATION validation OPTIONAL)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/native-test-results.txt DESTINATION validation OPTIONAL)
install(DIRECTORY third_party/ DESTINATION third_party)
install(DIRECTORY resources/icons/ DESTINATION artwork)
qt_generate_deploy_qml_app_script(TARGET StrataCopy OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
